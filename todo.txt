# 1. État général du namespace gitlab
kubectl get all -n gitlab -o wide

# 2. Détail du pod (événements, raison du redémarrage/crash éventuel)
kubectl describe pod -n gitlab -l app=gitlab | tail -80

# 3. Logs du conteneur (le reconfigure omnibus affiche sa progression ici)
kubectl logs -n gitlab -l app=gitlab --tail=80

# 4. Vérifier si le pod a été tué par manque de mémoire (OOMKilled)
kubectl get pod -n gitlab -l app=gitlab -o jsonpath='{.items[0].status.containerStatuses[0].lastState}'

# 5. Mémoire/CPU réellement consommés par le pod
kubectl top pod -n gitlab

# 6. Mémoire/CPU de la VM elle-même (pour voir si c'est la VM qui sature)
free -h && uptime
